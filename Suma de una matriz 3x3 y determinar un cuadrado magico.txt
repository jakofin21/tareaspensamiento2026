#Determinar cuadrado magico:
matriz = []

print("Ingrese la matriz 3x3:")

for i in range(3):
    fila = []
    for j in range(3):
        print("Elemento fila", i + 1, "columna", j + 1)
        valor = int(input())
        fila.append(valor)
    matriz.append(fila)

print("Matriz final:")
for fila in matriz:
    print(fila)

constante = matriz[0][0] + matriz[0][1] + matriz[0][2]

es_magico = True

for i in range(3):
    if matriz[i][0] + matriz[i][1] + matriz[i][2] != constante:
        es_magico = False

for j in range(3):
    if matriz[0][j] + matriz[1][j] + matriz[2][j] != constante:
        es_magico = False

if matriz[0][0] + matriz[1][1] + matriz[2][2] != constante:
    es_magico = False

if matriz[0][2] + matriz[1][1] + matriz[2][0] != constante:
    es_magico = False

if es_magico:
    print("¡Felicidades! Es un cuadrado mágico")
else:
    print("No es un cuadrado mágico")
